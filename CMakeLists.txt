cmake_minimum_required(VERSION 3.0.2)
project(ehpr_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS lio_sam roscpp roslib sensor_msgs nav_msgs geometry_msgs
  std_msgs std_srvs diagnostic_msgs message_filters pcl_ros pcl_conversions message_generation)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree registration)
find_package(yaml-cpp REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(TBB REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLOG REQUIRED libglog)
find_package(PythonInterp 3 REQUIRED)
set(STAGE3_UPSTREAM ${CMAKE_CURRENT_BINARY_DIR}/ehpr_stage3)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  patches/ehpr_stage3.patch scripts/prepare_stage3_upstream.py third_party/EHPR_STAGE3_SOURCE.json)
execute_process(COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_stage3_upstream.py
  ${CMAKE_CURRENT_SOURCE_DIR}/third_party/EHPR ${STAGE3_UPSTREAM}
  ${CMAKE_CURRENT_SOURCE_DIR}/patches/ehpr_stage3.patch RESULT_VARIABLE PATCH_RESULT)
if(NOT PATCH_RESULT EQUAL 0)
  message(FATAL_ERROR "Cannot prepare pinned Stage 3 upstream build copy")
endif()
add_message_files(FILES RobotSubmap.msg InterRobotMatch.msg MapAlignment.msg)
generate_messages(DEPENDENCIES std_msgs sensor_msgs geometry_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES ehpr_submap_core stage3_matcher stage3_io
  CATKIN_DEPENDS roscpp sensor_msgs nav_msgs geometry_msgs std_msgs message_runtime
  DEPENDS EIGEN3 PCL OpenCV)
include_directories(include third_party/EHPR/include ${catkin_INCLUDE_DIRS}
  ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(ehpr_submap_core src/submap_session.cpp)
target_link_libraries(ehpr_submap_core ${PCL_LIBRARIES})
add_library(stage3_matcher src/matcher.cpp src/visibility.cpp src/alignment_tracker.cpp src/partial_registration.cpp src/contour_registration.cpp src/advanced_matcher.cpp src/session_map.cpp src/submap_fusion.cpp src/recent_cloud_buffer.cpp ${STAGE3_UPSTREAM}/src/ehpr/desc_mng.cpp
  ${STAGE3_UPSTREAM}/src/ehpr/desc_db.cpp ${STAGE3_UPSTREAM}/src/ehpr/AlignRansac2D.cpp)
target_include_directories(stage3_matcher BEFORE PRIVATE ${STAGE3_UPSTREAM}/include
  ${STAGE3_UPSTREAM}/thirdparty ${OpenCV_INCLUDE_DIRS})
target_include_directories(stage3_matcher PUBLIC ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(stage3_matcher PRIVATE SAVE_MID_FILE=0 SRRG_HBST_HAS_EIGEN
  SRRG_HBST_HAS_OPENCV SRRG_MERGE_DESCRIPTORS)
target_compile_options(stage3_matcher PRIVATE -O2)
target_include_directories(stage3_matcher PRIVATE ${GLOG_INCLUDE_DIRS})
target_link_libraries(stage3_matcher ${PCL_LIBRARIES} ${OpenCV_LIBS} TBB::tbb ${GLOG_LIBRARIES} yaml-cpp)
add_library(stage3_io src/matching_io.cpp)
file(READ ${STAGE3_UPSTREAM}/STAGE3_PATCH.json STAGE3_SOURCE_RECORD_JSON)
configure_file(cmake/stage3_source_record.h.in generated/stage3_source_record.h @ONLY)
target_include_directories(stage3_io PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(stage3_io stage3_matcher yaml-cpp)
add_executable(stage3_match_cli src/match_cli.cpp)
target_link_libraries(stage3_match_cli stage3_io)
add_executable(stage3_matching_node src/matching_node.cpp)
add_dependencies(stage3_matching_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(stage3_matching_node stage3_io ${catkin_LIBRARIES})
add_executable(stage2_submap_node src/submap_node.cpp)
add_dependencies(stage2_submap_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(stage2_submap_node ehpr_submap_core ${catkin_LIBRARIES} yaml-cpp)
if(EHPR_VALIDATION_DIR)
  foreach(target stage3_matcher stage3_io stage3_match_cli stage3_matching_node)
    set_target_properties(${target} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${EHPR_VALIDATION_DIR}"
      RUNTIME_OUTPUT_DIRECTORY "${EHPR_VALIDATION_DIR}")
  endforeach()
endif()
catkin_install_python(PROGRAMS scripts/stage2_verify_exports.py scripts/stage2_ros_test.py
  scripts/stage3_verify_results.py scripts/stage3_truth_check.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS ehpr_submap_core stage2_submap_node stage3_matcher stage3_io stage3_match_cli stage3_matching_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY launch config patches DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES ${STAGE3_UPSTREAM}/STAGE3_PATCH.json DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md UPSTREAM.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES scripts/alignment_verify.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES third_party/EHPR_SOURCE.json DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/third_party)
install(FILES third_party/EHPR_STAGE3_SOURCE.json DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/third_party)
install(DIRECTORY third_party/EHPR/include/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/third_party/EHPR/include)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_matcher tests/test_matcher.cpp tests/test_partial_registration.cpp tests/test_advanced.cpp tests/test_alignment_safety.cpp)
  if(TARGET test_matcher)
    target_include_directories(test_matcher BEFORE PRIVATE ${STAGE3_UPSTREAM}/include ${STAGE3_UPSTREAM}/thirdparty)
    target_compile_definitions(test_matcher PRIVATE SAVE_MID_FILE=0 SRRG_HBST_HAS_EIGEN SRRG_HBST_HAS_OPENCV SRRG_MERGE_DESCRIPTORS)
    target_link_libraries(test_matcher stage3_io)
    if(EHPR_VALIDATION_DIR)
      set_target_properties(test_matcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${EHPR_VALIDATION_DIR}")
    endif()
  endif()
  catkin_add_gtest(test_submap_session tests/test_submap_session.cpp)
  if(TARGET test_submap_session)
    target_link_libraries(test_submap_session ehpr_submap_core ${PCL_LIBRARIES})
  endif()
endif()
